# Dependencies for the multi-cloud dispatcher + remote-finalize pipeline.
#
# Installed into scripts/.venv by `make scripts-venv`.  Kept minimal so
# the dispatcher's surface area stays small and auditable.
#

# remote run results (X25519 + ML-KEM-768 + ChaCha20-Poly1305).

# Backblaze B2 client.  rclone is preferred for the pod-side upload
# (single static binary, no Python deps), but the official b2-sdk-python

# Unique-by-construction staging-dir suffixes (adjective-noun slugs like

# dispatchers in the same UTC second silently shared one staging dir
# under mkdir(exist_ok=True)).  Pure Python, ~10^7 unique slug pairs.
coolname>=2.2,<3.0

# Testing.
pytest>=8.0,<9.0
